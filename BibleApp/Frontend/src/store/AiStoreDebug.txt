  removeVerseFromContext: async (verseToRemove) => {
    console.log('🗑️ removeVerseFromContext llamado con:', verseToRemove);
    const { verseToExplain, currentConversationId } = get();
    
    console.log('📊 Estado actual:', {
      verseToExplain: verseToExplain.length,
      currentConversationId
    });
    
    const updatedVerses = verseToExplain.filter(v => 
      !(v.bookName === verseToRemove.bookName &&
        v.chapterNumber === verseToRemove.chapterNumber &&
        v.verseNumber === verseToRemove.verseNumber)
    );
    
    console.log('✂️ Versículos después de filtrar:', updatedVerses.length);
    set({ verseToExplain: updatedVerses });
    
    if (currentConversationId) {
      console.log('💾 Actualizando DB para conversación:', currentConversationId);
      await get().updateVerseContextInDB(updatedVerses);
    } else {
      console.log('⚠️ No hay conversación activa, no se actualiza DB');
    }
  },

  updateVerseContextInDB: async (updatedVerses) => {
    console.log('🔄 updateVerseContextInDB iniciado');
    console.log('📝 Versículos actualizados a mantener:', updatedVerses.length);
    
    const { currentConversationId } = get();
    
    if (!currentConversationId) {
      console.warn('⚠️ No hay conversación activa para actualizar');
      return;
    }

    console.log('🔍 Buscando mensajes de conversación:', currentConversationId);

    const { data: messages, error: fetchError } = await supabase
      .from('conversation_messages')
      .select('id, verse_context')
      .eq('conversation_id', currentConversationId);

    if (fetchError) {
      console.error('❌ Error al obtener mensajes:', fetchError);
      return;
    }

    console.log('📨 Mensajes encontrados:', messages?.length || 0);

    if (!messages || messages.length === 0) {
      console.log('⚠️ No hay mensajes para actualizar');
      return;
    }

    const messagesToUpdate = messages.filter(msg => msg.verse_context && Array.isArray(msg.verse_context));
    console.log('📋 Mensajes con verse_context:', messagesToUpdate.length);

    const updates = messagesToUpdate.map(msg => {
      console.log('🔍 Procesando mensaje:', msg.id);
      console.log('  - Contexto original:', msg.verse_context.length, 'versículos');
      
      const filteredContext = msg.verse_context.filter(verse => 
        updatedVerses.some(v => 
          v.bookName === verse.bookName &&
          v.chapterNumber === verse.chapterNumber &&
          v.verseNumber === verse.verseNumber
        )
      );

      console.log('  - Contexto filtrado:', filteredContext.length, 'versículos');
      console.log('  - Se guardará como:', filteredContext.length > 0 ? 'array' : 'null');

      return supabase
        .from('conversation_messages')
        .update({ verse_context: filteredContext.length > 0 ? filteredContext : null })
        .eq('id', msg.id);
    });

    console.log('💾 Ejecutando', updates.length, 'actualizaciones...');

    const results = await Promise.all(updates);
    
    const errors = results.filter(r => r.error);
    if (errors.length > 0) {
      console.error('❌ Errores en actualizaciones:', errors);
    } else {
      console.log(`✅ Contexto actualizado exitosamente en ${updates.length} mensajes`);
    }
  },
